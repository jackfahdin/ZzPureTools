cmake_minimum_required(VERSION 3.23)
project(GaugeFrameTimeline LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "This diagnostic probe observes the Linux XCB backend")
endif()
find_package(ZzPureToolsFrame CONFIG REQUIRED)
add_executable(gauge-timeline probe.cpp)
target_compile_features(gauge-timeline PRIVATE cxx_std_20)
target_link_libraries(gauge-timeline PRIVATE Zz::FluentUI ${CMAKE_DL_LIBS})
set_target_properties(gauge-timeline PROPERTIES ENABLE_EXPORTS ON)
